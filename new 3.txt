class ChatApp {
    constructor() {
        this.currentConversationId = null;
        this.currentOtherUserId = null;
        this.currentOtherUserName = null;
        this.pollingInterval = null;
        this.lastMessageId = 0;
        this.isLoading = false;
        this.isTyping = false;
        this.typingTimeout = null;
        this.emojiPickerVisible = false;
        this.uploadedFiles = [];
        this.isRecording = false;
        this.mediaRecorder = null;
        this.audioChunks = [];
        this.isDarkTheme = localStorage.getItem('chatAppTheme') === 'dark';
        
        this.init();
    }

    init() {
        console.log('ChatApp initialized with all features');
        this.loadConversations();
        this.setupEventListeners();
        this.startPolling();
        this.applyTheme();
        this.initEmojiPicker();
    }

    setupEventListeners() {
        // Message input events
        const messageInput = document.getElementById('messageInput');
        if (messageInput) {
            messageInput.addEventListener('input', () => {
                this.handleTyping();
                this.toggleSendButton();
            });

            messageInput.addEventListener('keydown', (e) => {
                if (e.key === 'Enter' && !e.shiftKey) {
                    e.preventDefault();
                    this.sendMessage();
                }
            });
        }

        // File input
        const fileInput = document.getElementById('fileInput');
        if (fileInput) {
            fileInput.addEventListener('change', (e) => {
                this.handleFileUpload(e.target.files);
            });
        }

        // Search functionality
        const searchUsers = document.getElementById('searchUsers');
        if (searchUsers) {
            searchUsers.addEventListener('input', (e) => {
                this.searchConversations(e.target.value);
            });
        }

        const modalSearch = document.getElementById('modalSearch');
        if (modalSearch) {
            modalSearch.addEventListener('input', (e) => {
                this.searchUsers(e.target.value);
            });
        }

        // Click outside modals and emoji picker
        document.addEventListener('click', (e) => {
            const modal = document.getElementById('newChatModal');
            if (e.target === modal) {
                this.hideModal();
            }
            
            // Close emoji picker when clicking outside
            const emojiPicker = document.getElementById('emojiPicker');
            if (emojiPicker && !emojiPicker.contains(e.target) && 
                !e.target.closest('.input-action-btn')) {
                this.hideEmojiPicker();
            }
        });

        // Escape key to close modals
        document.addEventListener('keydown', (e) => {
            if (e.key === 'Escape') {
                this.hideAllModals();
                this.hideEmojiPicker();
            }
        });

        // Theme toggle
        const themeBtn = document.querySelector('[onclick="toggleTheme()"]');
        if (themeBtn) {
            themeBtn.addEventListener('click', () => this.toggleTheme());
        }

        // Online status simulation
        setInterval(() => {
            this.updateOnlineStatus();
        }, 30000);
    }

    // ==================== THEME MANAGEMENT ====================
    toggleTheme() {
        this.isDarkTheme = !this.isDarkTheme;
        document.body.classList.toggle('dark-theme', this.isDarkTheme);
        localStorage.setItem('chatAppTheme', this.isDarkTheme ? 'dark' : 'light');
        this.showNotification(`Theme changed to ${this.isDarkTheme ? 'dark' : 'light'} mode`);
    }

    applyTheme() {
        if (this.isDarkTheme) {
            document.body.classList.add('dark-theme');
        }
    }

    async startRecording() {
        try {
            const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
            this.mediaRecorder = new MediaRecorder(stream);
            this.audioChunks = [];

            this.mediaRecorder.ondataavailable = (event) => {
                this.audioChunks.push(event.data);
            };

            this.mediaRecorder.onstop = async () => {
                const audioBlob = new Blob(this.audioChunks, { type: 'audio/wav' });
                await this.sendVoiceMessage(audioBlob);
                stream.getTracks().forEach(track => track.stop());
            };

            this.mediaRecorder.start();
            this.isRecording = true;
            this.updateVoiceRecordUI(true);
            
        } catch (error) {
            console.error('Error starting recording:', error);
            this.showError('Cannot access microphone');
        }
    }

    stopRecording() {
        if (this.mediaRecorder && this.isRecording) {
            this.mediaRecorder.stop();
            this.isRecording = false;
            this.updateVoiceRecordUI(false);
        }
    }

    updateVoiceRecordUI(recording) {
        const voiceBtn = document.querySelector('[onclick="toggleVoiceMessage()"]');
        const micIcon = voiceBtn?.querySelector('i');
        
        if (recording) {
            if (micIcon) micIcon.className = 'fas fa-stop';
            voiceBtn.style.color = '#e74c3c';
            this.showNotification('Recording... Click stop to send', 'info');
        } else {
            if (micIcon) micIcon.className = 'fas fa-microphone';
            voiceBtn.style.color = '';
        }
    }

    renderMessages(messages) {
        const container = document.getElementById('messagesContainer');
        if (!container) return;
        
        const currentUserId = <?php echo $user_id; ?>;
        
        if (!messages || messages.length === 0) {
            container.innerHTML = this.getEmptyStateHTML();

    groupMessagesByDateAndSender(messages) {
        const groups = {};
        
        messages.forEach(message => {
            const date = new Date(message.created_at).toDateString();
            if (!groups[date]) {
                groups[date] = [];
            }
            
            const lastGroup = groups[date][groups[date].length - 1];
            if (lastGroup && lastGroup.sender_id === message.sender_id) {
                lastGroup.messages.push(message);
            } else {
                groups[date].push({
                    sender_id: message.sender_id,
                    sender_name: message.sender_name,
                    messages: [message]
                });
            }
        });
        
        return groups;
    }

    renderGroupedMessages(groupedMessages, currentUserId) {
        return Object.keys(groupedMessages).map(date => `

            ${groupedMessages[date].map(group => `
                <div class="message-group ${group.sender_id === currentUserId ? 'own-group' : 'other-group'}">
                    ${group.messages.map((message, index) => `
                        <div class="message ${message.sender_id === currentUserId ? 'sent' : 'received'}"
                             oncontextmenu="event.preventDefault(); window.chatApp.showMessageMenu(event, ${message.id})">
                            ${index === 0 && group.sender_id !== currentUserId ? `

                            <div class="message-bubble">

    // ==================== UTILITY FUNCTIONS ====================
    escapeHtml(unsafe) {
        if (!unsafe) return '';
        return unsafe
            .replace(/&/g, "&amp;")
            .replace(/</g, "&lt;")
            .replace(/>/g, "&gt;")
            .replace(/"/g, "&quot;")
            .replace(/'/g, "&#039;");
    }

    formatTime(timestamp) {
        const date = new Date(timestamp);
        return date.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
    }

    formatDate(dateString) {
        const date = new Date(dateString);
        const today = new Date();
        const yesterday = new Date(today);
        yesterday.setDate(yesterday.getDate() - 1);
        
        if (date.toDateString() === today.toDateString()) {
            return 'Today';
        } else if (date.toDateString() === yesterday.toDateString()) {
            return 'Yesterday';
        } else {
            return date.toLocaleDateString([], { month: 'short', day: 'numeric', year: 'numeric' });
        }
    }

    formatRelativeTime(timestamp) {
        if (!timestamp) return '';
        
        const date = new Date(timestamp);
        const now = new Date();
        const diffMs = now - date;
        const diffMins = Math.floor(diffMs / 60000);
        const diffHours = Math.floor(diffMs / 3600000);
        const diffDays = Math.floor(diffMs / 86400000);
        
        if (diffMins < 1) return 'Just now';
        if (diffMins < 60) return `${diffMins}m`;
        if (diffHours < 24) return `${diffHours}h`;
        if (diffDays < 7) return `${diffDays}d`;
        return date.toLocaleDateString([], { month: 'short', day: 'numeric' });
    }

    formatFileSize(bytes) {
        if (!bytes) return '0 Bytes';
        const k = 1024;
        const sizes = ['Bytes', 'KB', 'MB', 'GB'];
        const i = Math.floor(Math.log(bytes) / Math.log(k));
        return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
    }

    async sendMessage() {
        const messageInput = document.getElementById('messageInput');
        const messageText = messageInput ? messageInput.value.trim() : '';

        if (!messageText) return;

        const sendButton = document.getElementById('sendButton');
        if (sendButton) sendButton.disabled = true;

        try {
            const response = await fetch('ajax/send_message.php', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                },
                body: JSON.stringify({
                    conversation_id: this.currentConversationId,
                    message_text: messageText
                })
            });

            const data = await response.json();

            if (data.success) {
                if (messageInput) messageInput.value = '';
                this.addMessageToView(data.message);
                this.scrollToBottom();
                this.toggleSendButton();
                this.loadConversations();
            } else {
                this.showError(data.message || 'Failed to send message');
            }
        } catch (error) {
            console.error('Error sending message:', error);
            this.showError('Network error sending message');
        } finally {
            if (sendButton) sendButton.disabled = false;
        }
    }

    showError(message) {
        this.showNotification(message, 'error');
    }

    toggleSendButton() {
        const messageInput = document.getElementById('messageInput');
        const sendButton = document.getElementById('sendButton');
        if (sendButton && messageInput) {
            sendButton.disabled = !messageInput.value.trim();
        }
    }

    scrollToBottom() {
        const container = document.getElementById('messagesContainer');
        if (container) {
            container.scrollTop = container.scrollHeight;
        }
    }

    startPolling() {
        this.pollingInterval = setInterval(() => {
            if (this.currentConversationId) {
                this.checkNewMessages();
            }
            this.loadConversations();
        }, 3000);
    }

    async checkNewMessages() {
        try {
            const response = await fetch(`ajax/get_messages.php?conversation_id=${this.currentConversationId}&last_id=${this.lastMessageId}`);
            const data = await response.json();
            
            if (data.success && data.messages.length > 0) {
                data.messages.forEach(message => {
                    this.addMessageToView(message);
                });
                this.scrollToBottom();
                this.lastMessageId = Math.max(this.lastMessageId, ...data.messages.map(m => m.id));
            }
        } catch (error) {
            console.error('Error checking new messages:', error);
        }
    }

    // ... [Keep all other existing methods]
}
